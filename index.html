My Java Project

Event Ticket Sales Analyzer

Project: Final Performance Task

Programming Language: Java


Project Description

Event Ticket Sales Analyzer stores the number of tickets sold for 10 events in an int array, accepting nonnegative integers. It calculates total and average ticket sales, identifies the highest-selling event, and counts events selling at least 100 tickets.


Java Source Code

import java.util.Scanner;

public class TicketSalesAnalyzer {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        int[] tickets = new int[10];
        int total = 0;
        int highest = 0;
        int count = 0;
        double average;

        for (int i = 0; i < 10; i++) {

            System.out.print("Enter tickets sold for event "
                    + (i + 1) + ": ");

            tickets[i] = input.nextInt();

            while (tickets[i] < 0) {
                System.out.print(
                    "Invalid number. Enter a nonnegative number: ");
                tickets[i] = input.nextInt();
            }

            total += tickets[i];

            if (tickets[i] > highest) {
                highest = tickets[i];
            }

            if (tickets[i] >= 100) {
                count++;
            }
        }

        average = total / 10.0;

        System.out.println("\n--- Event Ticket Sales Report ---");
        System.out.println("Total tickets sold: " + total);
        System.out.println("Average ticket sales: " + average);
        System.out.println("Highest tickets sold: " + highest);
        System.out.println(
            "Events selling at least 100 tickets: " + count);

        System.out.println("\nHighest-selling event(s):");

        for (int i = 0; i < 10; i++) {
            if (tickets[i] == highest) {
                System.out.println("Event " + (i + 1));
            }
        }

        input.close();
    }
}
